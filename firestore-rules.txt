rules_version='2';
service cloud.firestore{match /databases/{d}/documents{
 function admin(){return request.auth!=null&&request.auth.token.email=='mmbnb2022@gmail.com';}
 function me(){return request.auth!=null;}
 match /users/{u}{
  allow read:if me()&&(request.auth.uid==u||admin());
  allow create:if request.auth.uid==u&&request.resource.data.balance==0;
  allow update:if admin()||(request.auth.uid==u&&((request.resource.data.diff(resource.data).affectedKeys().hasOnly(['balance'])&&request.resource.data.balance<resource.data.balance&&request.resource.data.balance>=0)||(request.resource.data.diff(resource.data).affectedKeys().hasOnly(['photo'])&&request.resource.data.photo.size()<40000)));}
 match /jobs/{j}{allow read:if me();allow write:if admin();}
 match /notices/{n}{allow read:if me();allow write:if admin();}
 match /submissions/{s}{allow read:if me()&&(resource.data.uid==request.auth.uid||admin());
  allow create:if me()&&request.resource.data.uid==request.auth.uid&&request.resource.data.status=='pending';
  allow update,delete:if admin();}
 match /withdrawals/{w}{allow read:if me()&&(resource.data.uid==request.auth.uid||admin());
  allow create:if me()&&request.resource.data.uid==request.auth.uid&&request.resource.data.status=='pending';
  allow update,delete:if admin();}
}}
